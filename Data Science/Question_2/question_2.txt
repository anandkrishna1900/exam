# Ex. No: 2 - To write an R program to count the even and odd numbers in a vector.

numbers <- c(12, 5, 7, 8, 14, 3, 9, 10)
even_count <- 0
odd_count <- 0
for (num in numbers) {
  if (num %% 2 == 0) {
    even_count <- even_count + 1
  } else {
    odd_count <- odd_count + 1
  }
}
cat("Even numbers:", even_count, "\n")
cat("Odd numbers:", odd_count, "\n")
