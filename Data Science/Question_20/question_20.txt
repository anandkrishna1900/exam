# 20. Implementation of descriptive statistics in R - mean, median, range and variance

x <- c(10, 2, 30, 4, 5, 6, 70, 8, 9, 10)
y <- c(1, 2, 3, 4, 5, NA)

# Mean of x
mean(x)
# Mean of x with trim attribute
mean(x, trim = 0.2)
# Mean of y with na.rm attribute
mean(y, na.rm = TRUE)

x <- c(10, 2, 30, 2, 5, 8, 70, 8, 9, 2)
y <- c(3, 2, 3, 4, 5, NA)

# Median of x
median(x)
# Median of y with na.rm attribute
median(y, na.rm = TRUE)

x <- c(10, 2, 30, 4, 5, 6, 70, 8, 9, 10)
y <- c(1, 2, 3, 4, 5, NA)

# Range
range(x)
range(y, na.rm = FALSE)

x <- c(10, 2, 30, 2, 5, 8, 70, 8, 9, 2)
y <- c(3, 2, 3, 4, 5, NA)

# Variance of x
var(x)
# Variance of y with na.rm attribute
var(y, na.rm = TRUE)
